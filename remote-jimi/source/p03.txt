local function isProxyText(text)
    text = strip(text or "")
    return text:find("%[" .. TAG .. "%]") or text:find(TAG, 1, true)
end

local function numColor(n)
    n = tonumber(n) or 0
    if n == 0 then return "`4" end
    if n % 2 == 0 then return "`2" end
    return "`4"
end

local function calcWheel(n)
    n = tonumber(n) or 0
    local result = 0
    if rouletteMode == "QEME" then
        result = n % 10
        if result == 0 then return result, "QEME", " `6x3" end
        return result, "QEME", ""
    end
    local sum = 0
    for c in tostring(n):gmatch("%d") do sum = sum + (tonumber(c) or 0) end
    result = sum % 10
    if rouletteMode == "LEME" then
        if result == 0 then return result, "LEME", " `6x4" end
        if result == 1 then return result, "LEME", " `6x3" end
        if result == 9 then return result, "LEME", " `4AUTO LOSE" end
        return result, "LEME", ""
    end
    if result == 0 then return result, "REME", " `6x3" end
    return result, "REME", ""
end

local function parseWheel(text)
    local clean = strip(text or ""):gsub("\r", " "):gsub("\n", " "):gsub("%s+", " ")
    clean = trim(clean)
    if clean == "" or isProxyText(clean) then return nil end
    if not clean:lower():find("spun the wheel", 1, true) then return nil end
    local p, n = clean:match("^(.-)%s+spun%s+the%s+wheel%s+and%s+got%s+(%d+)")
    if not p or not n then return nil end
    p = trim(p)
    p = p:gsub("^%[[^%]]-%]%s*", "")
    p = p:gsub("^%d%d:%d%d:%d%d%s*", "")
    if nickReal ~= "" and nickName ~= "" then
        local low = p:lower()
        if low:find(nickName:lower(), 1, true) then p = nickName end
        if low:find(nickReal:lower(), 1, true) then p = nickName end
    end
    return {player = wheelNickName(p), number = tonumber(n)}
end

local function wheelLine(player, number, mode, result, extra)
    extra = tostring(extra or "")
    return "`w[ `2REAL `w] [`2" .. tostring(player or myName()) .. " `wspun the wheel and got " .. numColor(number) .. tostring(number) .. "`w!] [ `2" .. mode .. " `w] [ `2" .. tostring(result) .. " `w]" .. extra
end

local function emitWheel(parsed, netid)
    if not parsed then return false end
    local result, mode, extra = calcWheel(parsed.number)
    local pname = wheelNickName(parsed.player)
    local key = tostring(pname):lower() .. ":" .. tostring(parsed.number) .. ":" .. mode
    local now = ms()
    if recentSpin[key] and (now - recentSpin[key]) < 1700 then return true end
    recentSpin[key] = now
    local line = wheelLine(pname, parsed.number, mode, result, extra)
    push(wheelLogs, {time = stamp(), world = worldName(), text = strip(line)}, 140)
    sendConsole(prefix() .. line)
    if bubbleRoulette then bubble(line, netid) end
    return true
end

local function logDrop(text)
    push(dropLogs, {time = stamp(), world = worldName(), text = text}, 140)
end

local function confirmDropPacket(itemID, count)
    packet2("action|dialog_return\ndialog_name|drop_item\nitemID|"..tostring(itemID).."|\ncount|"..tostring(count))
end

local function onlyPendingDrop()
    local foundID, foundData, total = nil, nil, 0
    local now = ms()
    for id, data in pairs(pendingDrops) do
        if data and (now - (tonumber(data.ts) or 0)) <= 2500 then
            foundID, foundData = id, data
            total = total + 1
        end
    end
    if total == 1 then return foundID, foundData end
    return nil, nil
end

local function dropItemFast(itemID, count, label)
    itemID = tonumber(itemID) or 0
    count = tonumber(count) or 0
    label = label or lockInfo(itemID).short
    if count <= 0 then say("`4Invalid amount."); return true end
    if itemCount(itemID) < count then
        say("`4Not enough " .. tostring(label) .. " (`wNeed " .. count .. "`4, have " .. itemCount(itemID) .. ")")
        return true
    end
    pendingDrops[itemID] = {count = count, label = label, ts = ms(), sent = false}
    packet2("action|drop\nitemID|" .. tostring(itemID))
    runThread(function()
        sleep(45)
        local pd = pendingDrops[itemID]
        if pd and pd.count == count and not pd.sent then
            confirmDropPacket(itemID, count)
            pd.sent = true
        end
        sleep(1300)
        local last = pendingDrops[itemID]
        if last and last.count == count and (ms() - (last.ts or 0)) >= 1200 then pendingDrops[itemID] = nil end
    end)
    local msg = "Drop " .. tostring(count) .. "x " .. tostring(label)
    logDrop(msg)
    say("`2" .. msg)
    overlay("`9Dropped `2"..msg)
    return true
end
